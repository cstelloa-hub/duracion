Option Explicit

' === Columnas de la DB-Histórica CP (2) ===
Private Const DB_EMISION As Long = 1   ' A = Fecha de emisión
Private Const DB_VCTO As Long = 2, DB_EMIS As Long = 4, DB_SECT As Long = 7
Private Const DB_RATING As Long = 12, DB_ANIOS As Long = 14
Private Const DB_TASA As Long = 17, DB_TBASE As Long = 18
Private Const DB_SPREAD_PBS As Long = 20   ' T = Spread en pbs
Private Const DB_OFERTA As Long = 21       ' U = Oferta (monto)
Private Const DB_HOJA As String = "DB-Histórica CP (2)"
Private Const MAX_COMP As Long = 8

Sub BuscarComparables()
    Dim wb As Workbook: Set wb = ThisWorkbook
    Dim wsC As Worksheet, wsDB As Worksheet
    On Error GoTo fallo

    Set wsC = HojaComparables(wb)

    Dim inEmis As String, inSect As String, inRating As String
    Dim inPlazo As Double, inMonto As Double, carpeta As String
    inEmis = Trim(CStr(wsC.Range("B4").Value))
    inSect = Trim(CStr(wsC.Range("B5").Value))
    inRating = UCase(Trim(CStr(wsC.Range("B6").Value)))
    inPlazo = Val(wsC.Range("B7").Value)
    inMonto = Val(wsC.Range("B8").Value)
    carpeta = Trim(CStr(wsC.Range("B9").Value))

    If inPlazo <= 0 Or inMonto <= 0 Then
        MsgBox "Plazo (B7) y Monto (B8) son obligatorios.", vbExclamation: Exit Sub
    End If
    If Len(inEmis) = 0 And Len(inSect) = 0 Then
        MsgBox "Pon al menos Sector (B5) o Emisor (B4).", vbExclamation: Exit Sub
    End If

    On Error Resume Next
    Set wsDB = wb.Worksheets(DB_HOJA)
    On Error GoTo fallo
    If wsDB Is Nothing Then
        MsgBox "No encuentro la hoja '" & DB_HOJA & "'.", vbCritical: Exit Sub
    End If

    Dim lastDB As Long: lastDB = wsDB.Cells(wsDB.Rows.Count, DB_EMIS).End(xlUp).Row
    Dim dat As Variant
    dat = wsDB.Range(wsDB.Cells(2, 1), wsDB.Cells(lastDB, DB_OFERTA)).Value

    Dim res() As Variant, nRes As Long, nivel As String
    BuscaNivel dat, inEmis, "", inRating, inPlazo, 0.5, inMonto, 0.5, res, nRes
    nivel = "por EMISOR (±6m, monto ±50%)"
    If nRes < 3 Then
        BuscaNivel dat, "", inSect, inRating, inPlazo, 0.5, inMonto, 0.5, res, nRes
        nivel = "por SECTOR (±6m, monto ±50%)"
    End If
    If nRes < 3 Then
        BuscaNivel dat, "", inSect, inRating, inPlazo, 1#, inMonto, 0.5, res, nRes
        nivel = "por SECTOR (plazo ±1 año, monto ±50%)"
    End If
    If nRes < 3 Then
        BuscaNivel dat, "", inSect, inRating, inPlazo, 1#, inMonto, 0, res, nRes
        nivel = "por SECTOR (plazo ±1 año, sin filtro monto)"
    End If

    EscribeResultados wsC, res, nRes, nivel

    Dim saveMsg As String
    If Len(carpeta) > 0 Then
        If Right(carpeta, 1) <> "\" Then carpeta = carpeta & "\"
        If Len(Dir(carpeta, vbDirectory)) > 0 Then
            Dim nomE As String: nomE = LimpiaNombre(inEmis)
            If Len(nomE) = 0 Then nomE = LimpiaNombre(inSect)
            If Len(nomE) = 0 Then nomE = "SinNombre"
            Dim outPath As String
            outPath = carpeta & "Comparable_" & Format(Now, "yyyymmdd") & "_" & nomE & ".xlsx"
            Application.DisplayAlerts = False
            wsC.Copy
            ActiveWorkbook.SaveAs Filename:=outPath, FileFormat:=51
            ActiveWorkbook.Close SaveChanges:=False
            Application.DisplayAlerts = True
            saveMsg = vbCrLf & "Archivado: " & outPath
        Else
            saveMsg = vbCrLf & "Carpeta (B9) no existe: no se guardó copia."
        End If
    Else
        saveMsg = vbCrLf & "Sin carpeta en B9: no se guardó copia."
    End If

    MsgBox "Comparables: " & nRes & "  |  Match: " & nivel & saveMsg, vbInformation, "Comparables"
    Exit Sub
fallo:
    Application.DisplayAlerts = True
    MsgBox "ERROR: " & Err.Description, vbCritical, "Comparables"
End Sub

'------------------------------------------------------------------
Private Sub BuscaNivel(dat As Variant, emis As String, sect As String, rating As String, _
                       plazo As Double, tol As Double, monto As Double, tolM As Double, _
                       ByRef res() As Variant, ByRef nRes As Long)
    Dim i As Long, cnt As Long
    ReDim res(1 To 5000, 1 To 7)
    cnt = 0
    For i = 1 To UBound(dat, 1)
        Dim okE As Boolean, okS As Boolean, okR As Boolean, okP As Boolean, okM As Boolean
        okR = (Len(rating) = 0) Or (UCase(Trim(CStr(dat(i, DB_RATING)))) = rating)
        okP = IsNumeric(dat(i, DB_ANIOS))
        If okP Then okP = (Abs(CDbl(dat(i, DB_ANIOS)) - plazo) <= tol)
        okE = (Len(emis) = 0) Or (InStr(1, Trim(CStr(dat(i, DB_EMIS))), emis, vbTextCompare) > 0)
        okS = (Len(sect) = 0) Or (InStr(1, Trim(CStr(dat(i, DB_SECT))), sect, vbTextCompare) > 0)
        If tolM <= 0 Then
            okM = True
        ElseIf IsNumeric(dat(i, DB_OFERTA)) And monto > 0 Then
            okM = (Abs(CDbl(dat(i, DB_OFERTA)) - monto) <= monto * tolM)
        Else
            okM = True
        End If
        If okR And okP And okE And okS And okM Then
            If IsNumeric(dat(i, DB_SPREAD_PBS)) Then
                cnt = cnt + 1
                res(cnt, 1) = dat(i, DB_VCTO): res(cnt, 2) = dat(i, DB_EMIS)
                res(cnt, 3) = dat(i, DB_ANIOS): res(cnt, 4) = dat(i, DB_RATING)
                res(cnt, 5) = dat(i, DB_TBASE): res(cnt, 6) = CDbl(dat(i, DB_SPREAD_PBS))
                res(cnt, 7) = dat(i, DB_EMISION)
            End If
        End If
    Next i
    nRes = cnt
End Sub

'------------------------------------------------------------------
Private Sub EscribeResultados(ws As Worksheet, res As Variant, nRes As Long, nivel As String)
    ws.Range("A13:H300").Clear
    ws.Range("A13").Value = "Match: " & nivel & "   |   Comparables: " & nRes
    ws.Range("A13").Font.Bold = True

    Dim hd As Variant, c As Long
    hd = Array("Emisor", "Plazo", "Rating", "Vcto", "Tasa base", "Spread (pbs)")
    For c = 0 To UBound(hd): ws.Cells(15, c + 1).Value = hd(c): Next c

    Dim r As Long, k As Long, shown As Long
    r = 16: shown = 0
    Dim spreads() As Double
    If nRes > 0 Then ReDim spreads(1 To nRes)
    Dim tBaseHoy As Double, fMax As Double, fEmi As Double
    fMax = -1

    For k = 1 To nRes
        spreads(k) = CDbl(res(k, 6))   ' ya en pbs
        If IsNumeric(res(k, 5)) Then
            fEmi = -1
            If IsDate(res(k, 7)) Then fEmi = CDbl(CDate(res(k, 7)))
            If fEmi > fMax Then fMax = fEmi: tBaseHoy = CDbl(res(k, 5))
        End If
        If shown < MAX_COMP Then
            ws.Cells(r, 1).Value = res(k, 2)
            ws.Cells(r, 2).Value = res(k, 3)
            ws.Cells(r, 3).Value = res(k, 4)
            If IsDate(res(k, 1)) Then ws.Cells(r, 4).Value = CDate(res(k, 1))
            ws.Cells(r, 5).Value = res(k, 5)         ' tasa base (%)
            ws.Cells(r, 6).Value = res(k, 6)         ' spread (pbs) directo
            r = r + 1: shown = shown + 1
        End If
    Next k
    Dim ultComp As Long: ultComp = r - 1
    ws.Range(ws.Cells(16, 5), ws.Cells(ultComp, 5)).NumberFormat = "0.00%"
    ws.Range(ws.Cells(16, 6), ws.Cells(ultComp, 6)).NumberFormat = "0 ""pbs"""
    ws.Columns("D").NumberFormat = "dd/mm/yyyy"

    Dim baseR As Long: baseR = r + 1
    ws.Cells(baseR, 1).Value = "DISTRIBUCIÓN DEL SPREAD (pbs)": ws.Cells(baseR, 1).Font.Bold = True
    If nRes > 0 Then
        Dim wf As WorksheetFunction: Set wf = Application.WorksheetFunction
        Dim medi As Double, q1 As Double, q3 As Double, mn As Double, mx As Double
        mn = wf.Min(spreads): mx = wf.Max(spreads): medi = wf.Median(spreads)
        q1 = wf.Quartile(spreads, 1): q3 = wf.Quartile(spreads, 3)
        ws.Cells(baseR + 1, 1).Value = "Mínimo":  ws.Cells(baseR + 1, 2).Value = mn
        ws.Cells(baseR + 2, 1).Value = "Q1":      ws.Cells(baseR + 2, 2).Value = q1
        ws.Cells(baseR + 3, 1).Value = "Mediana": ws.Cells(baseR + 3, 2).Value = medi: ws.Cells(baseR + 3, 1).Font.Bold = True
        ws.Cells(baseR + 4, 1).Value = "Q3":      ws.Cells(baseR + 4, 2).Value = q3
        ws.Cells(baseR + 5, 1).Value = "Máximo":  ws.Cells(baseR + 5, 2).Value = mx
        ws.Range(ws.Cells(baseR + 1, 2), ws.Cells(baseR + 5, 2)).NumberFormat = "0 ""pbs"""

        Dim lr As Long: lr = baseR + 7
        ws.Cells(lr, 1).Value = "LECTURA FINAL": ws.Cells(lr, 1).Font.Bold = True
        ws.Cells(lr + 1, 1).Value = "Tasa base (comp. más reciente):"
        ws.Cells(lr + 1, 2).Value = tBaseHoy
        If fMax > 0 Then ws.Cells(lr + 1, 3).Value = "emisión: " & Format(CDate(fMax), "dd/mm/yyyy")
        ws.Cells(lr + 2, 1).Value = "Spread mediano:": ws.Cells(lr + 2, 2).Value = medi
        ws.Cells(lr + 3, 1).Value = "TASA SUGERIDA:": ws.Cells(lr + 3, 1).Font.Bold = True
        ws.Cells(lr + 3, 2).Value = tBaseHoy + medi / 10000    ' pbs -> %
        ws.Cells(lr + 3, 2).Font.Bold = True
        ws.Cells(lr + 4, 1).Value = "Banda (Q1–Q3):"
        ws.Cells(lr + 4, 2).Value = Format(tBaseHoy + q1 / 10000, "0.00%") & " – " & Format(tBaseHoy + q3 / 10000, "0.00%")
        ws.Cells(lr + 1, 2).NumberFormat = "0.00%"
        ws.Cells(lr + 2, 2).NumberFormat = "0 ""pbs"""
        ws.Cells(lr + 3, 2).NumberFormat = "0.00%"
    End If
    ws.Columns("A:F").AutoFit
    FormatoProfuturo ws, 15, 1, 6
End Sub

'------------------------------------------------------------------
Private Sub FormatoProfuturo(ws As Worksheet, ByVal filaHdr As Long, _
                             ByVal colIni As Long, ByVal colFin As Long)
    Dim ult As Long
    ult = ws.Cells(ws.Rows.Count, colIni).End(xlUp).Row
    If ult < filaHdr Then ult = filaHdr
    ws.Cells.Font.Name = "Arial"
    ws.Range(ws.Cells(filaHdr, colIni), ws.Cells(ult, colFin)).Font.Size = 8
    With ws.Range(ws.Cells(filaHdr, colIni), ws.Cells(filaHdr, colFin))
        .Interior.Color = RGB(212, 12, 12)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
    End With
    Dim rng As Range
    Set rng = ws.Range(ws.Cells(filaHdr, colIni), ws.Cells(ult, colFin))
    rng.Borders(xlEdgeLeft).LineStyle = xlNone
    rng.Borders(xlEdgeRight).LineStyle = xlNone
    rng.Borders(xlInsideVertical).LineStyle = xlNone
    rng.Borders(xlInsideHorizontal).LineStyle = xlNone
    With rng.Borders(xlEdgeTop): .LineStyle = xlContinuous: .Weight = xlThin: End With
    With rng.Borders(xlEdgeBottom): .LineStyle = xlContinuous: .Weight = xlThin: End With
End Sub

'------------------------------------------------------------------
Private Function HojaComparables(wb As Workbook) As Worksheet
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = wb.Worksheets("Comparables")
    On Error GoTo 0
    If ws Is Nothing Then
        Set ws = wb.Worksheets.Add(Before:=wb.Worksheets(1))
        ws.Name = "Comparables"
        ws.Range("A1").Value = "BUSCADOR DE COMPARABLES — EMISIÓN NUEVA"
        ws.Range("A1").Font.Bold = True
        ws.Range("A3").Value = "INPUTS": ws.Range("A3").Font.Bold = True
        ws.Range("A4").Value = "Emisor:"
        ws.Range("A5").Value = "Sector:"
        ws.Range("A6").Value = "Rating (opcional):"
        ws.Range("A7").Value = "Plazo (años) *:"
        ws.Range("A8").Value = "Monto (mn) *:"
        ws.Range("A9").Value = "Carpeta para guardar:"
        ws.Range("A11").Value = "(* obligatorio; y al menos Sector o Emisor)"
        ws.Columns("A").AutoFit
    End If
    Set HojaComparables = ws
End Function

'------------------------------------------------------------------
Private Function LimpiaNombre(s As String) As String
    Dim bad As Variant, x As Variant, t As String
    t = s
    bad = Array("\", "/", ":", "*", "?", """", "<", ">", "|", " ", ".")
    For Each x In bad: t = Replace(t, x, "_"): Next x
    LimpiaNombre = t
End Function
