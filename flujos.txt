Private Sub GenerarFlujos(wb As Workbook)
    Dim wsB As Worksheet, wsF As Worksheet
    On Error Resume Next
    Set wsB = wb.Worksheets("Base F0")
    On Error GoTo 0
    If wsB Is Nothing Then Exit Sub

    ' Acumula Val_total por SEMANA de vencimiento.
    ' Clave = lunes de esa semana (fecha), así ordena y agrupa solo.
    Dim d As Object: Set d = CreateObject("Scripting.Dictionary")
    Dim last As Long: last = wsB.Cells(wsB.Rows.Count, 8).End(xlUp).Row   ' H = Vcto
    Dim i As Long, vcto As Variant, monto As Double, lun As Date, clave As String
    For i = B_DAT To last
        vcto = wsB.Cells(i, 8).Value
        monto = 0
        If IsNumeric(wsB.Cells(i, 7).Value) Then monto = CDbl(wsB.Cells(i, 7).Value)  ' G = Monto (mn)
        If IsDate(vcto) Then
            lun = CDate(vcto) - (Weekday(CDate(vcto), vbMonday) - 1)   ' lunes de la semana
            clave = Format(lun, "yyyymmdd")
            If d.Exists(clave) Then d(clave) = d(clave) + monto Else d(clave) = monto
        End If
    Next i

    Set wsF = HojaFija(wb, "Flujos F0")
    wsF.Range("A1").Value = "FLUJOS POR SEMANA DE VENCIMIENTO (Val_total, mn)"
    wsF.Range("A1").Font.Bold = True
    wsF.Range("A3").Value = "Semana (lun–dom)"
    wsF.Range("B3").Value = "Monto a recibir (mn)"
    wsF.Range("C3").Value = "% del total"
    wsF.Range("A3:C3").Font.Bold = True

    Dim n As Long: n = d.Count
    If n = 0 Then Exit Sub
    Dim claves() As String: ReDim claves(1 To n)
    Dim k As Variant, j As Long: j = 0
    For Each k In d.Keys
        j = j + 1: claves(j) = CStr(k)
    Next k
    ' orden ascendente por fecha
    Dim a As Long, b As Long, tmp As String
    For a = 1 To n - 1
        For b = a + 1 To n
            If claves(b) < claves(a) Then tmp = claves(a): claves(a) = claves(b): claves(b) = tmp
        Next b
    Next a

    Dim total As Double
    For a = 1 To n: total = total + d(claves(a)): Next a

    Dim r As Long: r = 4, lunD As Date, domD As Date
    For a = 1 To n
        lunD = DateSerial(CInt(Left(claves(a), 4)), CInt(Mid(claves(a), 5, 2)), CInt(Right(claves(a), 2)))
        domD = lunD + 6
        wsF.Cells(r, 1).Value = Format(lunD, "dd/mm/yy") & " – " & Format(domD, "dd/mm/yy")
        wsF.Cells(r, 2).Value = d(claves(a))
        If total <> 0 Then wsF.Cells(r, 3).Value = d(claves(a)) / total
        r = r + 1
    Next a
    wsF.Cells(r, 1).Value = "TOTAL": wsF.Cells(r, 1).Font.Bold = True
    wsF.Cells(r, 2).Value = total: wsF.Cells(r, 2).Font.Bold = True
    If total <> 0 Then wsF.Cells(r, 3).Value = 1

    wsF.Columns("B").NumberFormat = "#,##0.00"
    wsF.Columns("C").NumberFormat = "0.0%"
    wsF.Columns("A:C").AutoFit
    FormatoInstitucional wsF, 3, 1, 3
End Sub
