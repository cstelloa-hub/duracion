' Crea la hoja "Rating F0" solo con las tablas de referencia
Private Sub CrearHojaRating(wb As Workbook)
    Dim ws As Worksheet: Set ws = HojaFija(wb, "Rating F0")
    ws.Range("A1").Value = "METODOLOGÍA DE CALIFICACIÓN — FONDO 0"
    ws.Range("A1").Font.Bold = True

    ws.Range("A3").Value = "CORTO PLAZO"
    ws.Range("A4").Value = "Rating": ws.Range("B4").Value = "Nota"
    ws.Range("A5").Value = "CPL 1+": ws.Range("B5").Value = 10
    ws.Range("A6").Value = "CPL 1":  ws.Range("B6").Value = 7
    ws.Range("A7").Value = "CPL 1-": ws.Range("B7").Value = 4

    ws.Range("D3").Value = "LARGO PLAZO"
    ws.Range("D4").Value = "Internacional": ws.Range("E4").Value = "Perú LP": ws.Range("F4").Value = "Nota"
    Dim t As Variant, i As Long
    t = Array( _
        Array("BBB-", "Soberanos", 10), _
        Array("BB+", "AAA", 7), _
        Array("BB", "AA+/AA", 6), _
        Array("BB-", "AA-", 5), _
        Array("B+", "A+", 4), _
        Array("B", "A/A-", 3), _
        Array("B-", "BBB+/BBB", 2))
    For i = 0 To UBound(t)
        ws.Cells(5 + i, 4).Value = t(i)(0)
        ws.Cells(5 + i, 5).Value = t(i)(1)
        ws.Cells(5 + i, 6).Value = t(i)(2)
    Next i

    FormatoHeadRating ws, "A3:B3": FormatoHeadRating ws, "A4:B4"
    FormatoHeadRating ws, "D3:F3": FormatoHeadRating ws, "D4:F4"
    ws.Columns("A:F").AutoFit
End Sub

'------------------------------------------------------------------
Private Sub FormatoHeadRating(ws As Worksheet, dir As String)
    With ws.Range(dir)
        .Interior.Color = RGB(212, 12, 12)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True: .Font.Name = "Arial": .Font.Size = 8
        .HorizontalAlignment = xlCenter
    End With
End Sub

' Escribe los 2 ratings ponderados (D9 corto, fila insertada largo) y tabla detalle abajo
Private Sub CalcularRatingEnCalculo(wb As Workbook)
    Dim wsB As Worksheet, wsK As Worksheet
    On Error Resume Next
    Set wsB = wb.Worksheets("Base F0")
    Set wsK = wb.Worksheets("Calculo F0")
    On Error GoTo 0
    If wsB Is Nothing Or wsK Is Nothing Then Exit Sub

    Dim last As Long: last = wsB.Cells(wsB.Rows.Count, 8).End(xlUp).Row
    Dim i As Long

    Dim sumNMc As Double, sumMc As Double, sumNMl As Double, sumMl As Double

    Dim nemos() As String, rInt() As String, rLoc() As String, esc() As String
    Dim notas() As Double, montos() As Double, nDatos As Long
    ReDim nemos(1 To 5000): ReDim rInt(1 To 5000): ReDim rLoc(1 To 5000)
    ReDim esc(1 To 5000): ReDim notas(1 To 5000): ReDim montos(1 To 5000)
    nDatos = 0

    For i = B_DAT To last
        Dim rating As String: rating = Trim(CStr(wsB.Cells(i, 12).Value))  ' L = Rating internacional
        Dim nemo As String: nemo = Trim(CStr(wsB.Cells(i, 1).Value))
        Dim monto As Double: monto = 0
        If IsNumeric(wsB.Cells(i, 7).Value) Then monto = CDbl(wsB.Cells(i, 7).Value)
        Dim escala As String, nota As Double, local As String
        NotaRating rating, escala, nota, local
        If escala <> "" And monto > 0 Then
            nDatos = nDatos + 1
            nemos(nDatos) = nemo: rInt(nDatos) = rating: rLoc(nDatos) = local
            esc(nDatos) = escala: notas(nDatos) = nota: montos(nDatos) = monto
            If escala = "Corto" Then
                sumNMc = sumNMc + nota * monto: sumMc = sumMc + monto
            Else
                sumNMl = sumNMl + nota * monto: sumMl = sumMl + monto
            End If
        End If
    Next i

    Dim pc As Double, pl As Double
    If sumMc > 0 Then pc = sumNMc / sumMc
    If sumMl > 0 Then pl = sumNMl / sumMl

    ' --- Corto en la fila 9 (etiqueta ya existe) ---
    wsK.Range("D9").Value = pc: wsK.Range("D9").NumberFormat = "0.00"
    wsK.Range("E9").Value = EquivCorto(pc)

    ' --- Largo: si la fila ya existe la actualiza, si no la inserta en la 10 ---
    Dim filaLargo As Long: filaLargo = 0
    Dim f As Long
    For f = 1 To 30
        If InStr(1, CStr(wsK.Cells(f, 1).Value), "Rating ponderado Largo", vbTextCompare) > 0 Then
            filaLargo = f: Exit For
        End If
    Next f
    If filaLargo = 0 Then
        wsK.Rows("10:10").Insert Shift:=xlDown
        filaLargo = 10
        wsK.Range("A10").Value = "Rating ponderado Largo"
    End If
    wsK.Cells(filaLargo, 4).Value = pl
    wsK.Cells(filaLargo, 4).NumberFormat = "0.00"
    wsK.Cells(filaLargo, 5).Value = EquivLargo(pl)

    ' --- Tabla detalle al final ---
    ' Borra un detalle anterior si existe (para no acumular al correr varias veces)
    Dim busca As Range
    Set busca = wsK.Columns(1).Find("DETALLE CALIFICACIÓN POR INSTRUMENTO", LookIn:=xlValues, LookAt:=xlWhole)
    If Not busca Is Nothing Then
        wsK.Range(wsK.Rows(busca.Row), wsK.Rows(wsK.Rows.Count)).ClearContents
    End If

    Dim r0 As Long: r0 = wsK.Cells(wsK.Rows.Count, 1).End(xlUp).Row + 3
    wsK.Cells(r0, 1).Value = "DETALLE CALIFICACIÓN POR INSTRUMENTO"
    wsK.Cells(r0, 1).Font.Bold = True
    Dim hr As Long: hr = r0 + 1
    wsK.Cells(hr, 1).Value = "Instrumento"
    wsK.Cells(hr, 2).Value = "Rating int."
    wsK.Cells(hr, 3).Value = "Rating local"
    wsK.Cells(hr, 4).Value = "Escala"
    wsK.Cells(hr, 5).Value = "Nota"
    wsK.Cells(hr, 6).Value = "Monto (mn)"
    wsK.Cells(hr, 7).Value = "Nota × Monto"

    Dim r As Long: r = hr + 1
    For i = 1 To nDatos
        wsK.Cells(r, 1).Value = nemos(i)
        wsK.Cells(r, 2).Value = rInt(i)
        wsK.Cells(r, 3).Value = rLoc(i)
        wsK.Cells(r, 4).Value = esc(i)
        wsK.Cells(r, 5).Value = notas(i)
        wsK.Cells(r, 6).Value = montos(i)
        wsK.Cells(r, 7).Value = notas(i) * montos(i)
        r = r + 1
    Next i
    wsK.Range(wsK.Cells(hr + 1, 6), wsK.Cells(r - 1, 7)).NumberFormat = "#,##0.00"

    With wsK.Range(wsK.Cells(hr, 1), wsK.Cells(hr, 7))
        .Interior.Color = RGB(212, 12, 12): .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True: .Font.Name = "Arial": .Font.Size = 8
    End With
    wsK.Range(wsK.Cells(hr, 1), wsK.Cells(r - 1, 7)).Font.Name = "Arial"
    wsK.Range(wsK.Cells(hr, 1), wsK.Cells(r - 1, 7)).Font.Size = 8
End Sub

'------------------------------------------------------------------
' Rating internacional -> escala, nota y rating local
Private Sub NotaRating(ByVal rating As String, ByRef escala As String, ByRef nota As Double, _
                       ByRef local As String)
    Dim s As String: s = UCase(Trim(rating))
    escala = "": nota = 0: local = ""
    If Left(s, 3) = "CPL" Then
        escala = "Corto"
        Dim c As String: c = Replace(Trim(Mid(s, 4)), " ", "")
        Select Case c
            Case "1+": nota = 10: local = "CPL 1+"
            Case "1":  nota = 7:  local = "CPL 1"
            Case "1-": nota = 4:  local = "CPL 1-"
        End Select
    ElseIf Left(s, 3) = "LPI" Or Left(s, 2) = "LP" Then
        escala = "Largo"
        Dim x As String
        If Left(s, 3) = "LPI" Then x = Trim(Mid(s, 4)) Else x = Trim(Mid(s, 3))
        x = Replace(x, " ", "")
        Select Case x
            Case "BBB-": nota = 10: local = "Soberanos"
            Case "BB+":  nota = 7:  local = "AAA"
            Case "BB":   nota = 6:  local = "AA+/AA"
            Case "BB-":  nota = 5:  local = "AA-"
            Case "B+":   nota = 4:  local = "A+"
            Case "B":    nota = 3:  local = "A/A-"
            Case "B-":   nota = 2:  local = "BBB+/BBB"
        End Select
    End If
End Sub

'------------------------------------------------------------------
Private Function EquivCorto(v As Double) As String
    Dim d10 As Double, d7 As Double, d4 As Double
    d10 = Abs(v - 10): d7 = Abs(v - 7): d4 = Abs(v - 4)
    If d10 <= d7 And d10 <= d4 Then
        EquivCorto = "CPL 1+"
    ElseIf d7 <= d4 Then
        EquivCorto = "CPL 1"
    Else
        EquivCorto = "CPL 1-"
    End If
End Function

'------------------------------------------------------------------
Private Function EquivLargo(v As Double) As String
    Dim notas As Variant, locales As Variant
    notas = Array(10, 7, 6, 5, 4, 3, 2)
    locales = Array("Soberanos", "AAA", "AA+/AA", "AA-", "A+", "A/A-", "BBB+/BBB")
    Dim mejor As Long, dmin As Double, i As Long
    mejor = 0: dmin = 1E+30
    For i = 0 To UBound(notas)
        If Abs(v - notas(i)) < dmin Then dmin = Abs(v - notas(i)): mejor = i
    Next i
    EquivLargo = locales(mejor)
End Function
